import { useState } from "react";
import { motion } from "framer-motion";

// Defines the shape of a product object
type Product = {
  kode: string;
  nama: string;
  warna: string;
  size: string;
  stok: string;
  harga: string;
  hpp: string;
};

// Defines the shape of a transaction object
type Transaction = {
  kode: string;
  nama: string;
  warna: string;
  size: string;
  stok: string;
  diskon: string;
  admin: string;
  keterangan: string;
};

// Utility function to format a number with thousand separators (e.g., 1000000 becomes 1.000.000)
// It first removes any non-digit characters to handle cases where the user has already typed a period.
const formatNumber = (value: string) => {
  if (!value) return "";
  const cleanValue = value.replace(/[^\d]/g, "");
  return cleanValue.replace(/\B(?=(\d{3})+(?!\d))/g, ".");
};

// Main component for the product management application
export default function App() {
  // State to manage the active tab ('product' or 'transaction')
  const [activeTab, setActiveTab] = useState<'product' | 'transaction'>('product');

  // State hooks for Product data and UI state
  const [products, setProducts] = useState<Product[]>([]);
  const [newProduct, setNewProduct] = useState<Product>({
    kode: "",
    nama: "",
    warna: "",
    size: "S",
    stok: "",
    harga: "",
    hpp: "",
  });
  const [editProductIndex, setEditProductIndex] = useState<number | null>(null);
  const [showProductForm, setShowProductForm] = useState(false);
  const [confirmProductPrint, setConfirmProductPrint] = useState<{ show: boolean; index: number | null }>({ show: false, index: null });
  const [confirmProductDelete, setConfirmProductDelete] = useState<{ show: boolean; index: number | null }>({ show: false, index: null });

  // State hooks for Transaction data and UI state
  const [transactions, setTransactions] = useState<Transaction[]>([]);
  const [newTransaction, setNewTransaction] = useState<Transaction>({
    kode: "",
    nama: "",
    warna: "",
    size: "S",
    stok: "",
    diskon: "",
    admin: "",
    keterangan: "",
  });
  const [editTransactionIndex, setEditTransactionIndex] = useState<number | null>(null);
  const [showTransactionForm, setShowTransactionForm] = useState(false);
  const [confirmTransactionDelete, setConfirmTransactionDelete] = useState<{ show: boolean; index: number | null }>({ show: false, index: null });

  const [infoMessage, setInfoMessage] = useState<{ show: boolean; message: string }>({ show: false, message: "" });

  // Function to save a new or edited product
  const handleSaveProduct = () => {
    // Basic form validation: check if all fields are filled
    if (!newProduct.kode || !newProduct.nama || !newProduct.warna || !newProduct.size || !newProduct.stok || !newProduct.harga || !newProduct.hpp) {
      setInfoMessage({ show: true, message: "Harap isi semua bidang." });
      return;
    }

    if (editProductIndex !== null) {
      // Logic for editing an existing product
      const updated = [...products];
      updated[editProductIndex] = newProduct;
      setProducts(updated);
    } else {
      // Logic for adding a new product
      setProducts([...products, newProduct]);
    }

    // Reset the form and hide the modal
    setNewProduct({ kode: "", nama: "", warna: "", size: "S", stok: "", harga: "", hpp: "" });
    setEditProductIndex(null);
    setShowProductForm(false);
  };

  // Function to handle product deletion
  const handleDeleteProduct = () => {
    if (confirmProductDelete.index !== null) {
      setProducts(products.filter((_, idx) => idx !== confirmProductDelete.index));
    }
    // Close the delete confirmation modal
    setConfirmProductDelete({ show: false, index: null });
  };

  // Function to handle printing of a product label
  const handlePrintProduct = () => {
    if (confirmProductPrint.index !== null) {
      const productToPrint = products[confirmProductPrint.index];
      const printWindow = window.open('', '_blank');
      if (printWindow) {
        // Create an HTML template for the product label
        printWindow.document.write(`
          <!DOCTYPE html>
          <html lang="en">
          <head>
              <meta charset="UTF-8">
              <meta name="viewport" content="width=device-width, initial-scale=1.0">
              <title>Product Label</title>
              <link rel="preconnect" href="https://fonts.googleapis.com">
              <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
              <link href="https://fonts.googleapis.com/css2?family=Poppins:wght@400;600;700&display=swap" rel="stylesheet">
              <style>
                  body {
                      font-family: 'Poppins', sans-serif;
                      display: flex;
                      justify-content: center;
                      align-items: center;
                      height: 100vh;
                      margin: 0;
                      background-color: #fff;
                  }
                  .label-container {
                      border: 1px solid #000;
                      padding: 10px;
                      width: 250px;
                      text-align: center;
                  }
                  .logo {
                      font-size: 24px;
                      font-weight: bold;
                      color: #d97706;
                      margin-bottom: 5px;
                  }
                  .product-name {
                      font-size: 18px;
                      font-weight: bold;
                      margin-bottom: 5px;
                  }
                  .product-details {
                      font-size: 14px;
                      margin-bottom: 5px;
                  }
                  .product-price {
                      font-size: 24px;
                      font-weight: bold;
                      color: #1e40af;
                  } 
                  .product-hpp {
                      font-size: 24px;
                      font-weight: bold;
                      color: #1e40af;
                  }
                  @media print {
                      body {
                          display: block;
                          height: auto;
                      }
                      .label-container {
                          border: none;
                          page-break-after: always;
                      }
                  }
              </style>
          </head>
          <body>
              <div class="label-container">
                  <div class="logo">Ocik Gallery</div>
                  <div class="product-name">${productToPrint.nama}</div>
                  <div class="product-details">Kode: ${productToPrint.kode}</div>
                  <div class="product-details">Warna: ${productToPrint.warna} | Ukuran: ${productToPrint.size} | Stok: ${productToPrint.stok}</div>
                  <div class="product-price">Rp ${formatNumber(productToPrint.harga)}</div>
                  <div class="product-price">Rp ${formatNumber(productToPrint.hpp)}</div>
              </div>
          </body>
          </html>
        `);
        printWindow.document.close();
        printWindow.print();
      } else {
        setInfoMessage({ show: true, message: "Gagal membuka jendela cetak. Periksa pengaturan pop-up browser Anda." });
      }
    }
    // Close the print confirmation modal
    setConfirmProductPrint({ show: false, index: null });
  };

  // Function to save a new or edited transaction
  const handleSaveTransaction = () => {
    if (!newTransaction.kode || !newTransaction.nama || !newTransaction.warna || !newTransaction.size || !newTransaction.stok || !newTransaction.diskon || !newTransaction.admin || !newTransaction.keterangan) {
      setInfoMessage({ show: true, message: "Harap isi semua bidang transaksi." });
      return;
    }

    if (editTransactionIndex !== null) {
      // Logic for editing an existing transaction
      const updated = [...transactions];
      updated[editTransactionIndex] = newTransaction;
      setTransactions(updated);
    } else {
      // Logic for adding a new transaction
      setTransactions([...transactions, newTransaction]);
    }

    setNewTransaction({
      kode: "",
      nama: "",
      warna: "",
      size: "S",
      stok: "",
      diskon: "",
      admin: "",
      keterangan: "",
    });
    setEditTransactionIndex(null);
    setShowTransactionForm(false);
  };

  // Function to handle transaction deletion
  const handleDeleteTransaction = () => {
    if (confirmTransactionDelete.index !== null) {
      setTransactions(transactions.filter((_, idx) => idx !== confirmTransactionDelete.index));
    }
    // Close the delete confirmation modal
    setConfirmTransactionDelete({ show: false, index: null });
  };


  return (
    <div className="p-6 min-h-screen bg-gray-100 font-poppins antialiased flex flex-col">
      <h1 className="text-3xl font-bold text-yellow-700 flex items-center gap-2 mb-6">
        <span className="text-4xl">🛒</span> Sistem Manajemen Stok
      </h1>

      {/* Tab Navigation */}
      <div className="flex bg-white p-2 rounded-2xl shadow-lg mb-6 max-w-lg mx-auto w-full">
        <button
          onClick={() => setActiveTab('product')}
          className={`flex-1 py-3 rounded-xl font-semibold transition-colors duration-300 ${
            activeTab === 'product'
              ? 'bg-yellow-500 text-white shadow-md'
              : 'text-gray-600 hover:bg-gray-100'
          }`}
        >
          Manajemen Produk
        </button>
        <button
          onClick={() => setActiveTab('transaction')}
          className={`flex-1 py-3 rounded-xl font-semibold transition-colors duration-300 ${
            activeTab === 'transaction'
              ? 'bg-yellow-500 text-white shadow-md'
              : 'text-gray-600 hover:bg-gray-100'
          }`}
        >
          Manajemen Transaksi
        </button>
      </div>

      {/* Conditional Rendering based on activeTab */}
      {activeTab === 'product' && (
        <div className="flex flex-col flex-grow">
          <div className="flex justify-between items-center mb-6">
            <h2 className="text-2xl font-bold text-yellow-700 flex items-center gap-2">
              <span className="text-3xl">🛍️</span> Produk
            </h2>
            {/* Button to add new product */}
            <button
              className="px-6 py-3 rounded-full bg-yellow-500 text-white font-semibold shadow-lg hover:bg-yellow-600 transition-transform transform hover:scale-105"
              onClick={() => {
                setShowProductForm(true);
                setEditProductIndex(null);
                setNewProduct({ kode: "", nama: "", warna: "", size: "S", stok: "", harga: "", hpp: "" });
              }}
            >
              <span className="flex items-center gap-2">
                <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M12 6v6m0 0v6m0-6h6m-6 0H6"></path></svg> Tambah Produk
              </span>
            </button>
          </div>

          {/* Product Table */}
          <div className="overflow-x-auto bg-white rounded-2xl shadow-xl flex-grow mb-12">
            <table className="w-full border-collapse">
              <thead className="bg-yellow-200 text-yellow-900 sticky top-0">
                <tr>
                  <th className="p-4 border-b-2 border-yellow-300">No.</th>
                  <th className="p-4 border-b-2 border-yellow-300">Kode</th>
                  <th className="p-4 border-b-2 border-yellow-300">Nama Produk</th>
                  <th className="p-4 border-b-2 border-yellow-300">Warna</th>
                  <th className="p-4 border-b-2 border-yellow-300">Ukuran</th>
                  <th className="p-4 border-b-2 border-yellow-300">Stok</th>
                  <th className="p-4 border-b-2 border-yellow-300">Harga</th>
                  <th className="p-4 border-b-2 border-yellow-300">HPP</th>
                  <th className="p-4 border-b-2 border-yellow-300">Action</th>
                </tr>
              </thead>
              <tbody>
                {products.map((p, i) => (
                  <motion.tr
                    key={i}
                    className="text-center even:bg-yellow-50 hover:bg-yellow-100 transition-colors"
                    initial={{ opacity: 0, y: 10 }}
                    animate={{ opacity: 1, y: 0 }}
                    transition={{ duration: 0.3, delay: i * 0.05 }}
                  >
                    <td className="p-3 border-b border-yellow-200">{i + 1}</td>
                    <td className="p-3 border-b border-yellow-200">{p.kode}</td>
                    <td className="p-3 border-b border-yellow-200">{p.nama}</td>
                    <td className="p-3 border-b border-yellow-200">{p.warna}</td>
                    <td className="p-3 border-b border-yellow-200">{p.size}</td>
                    <td className="p-3 border-b border-yellow-200">{p.stok}</td>
                    <td className="p-3 border-b border-yellow-200 font-semibold">Rp {formatNumber(p.harga)}</td>
                    <td className="p-3 border-b border-yellow-200 text-gray-600">Rp {formatNumber(p.hpp)}</td>
                    <td className="p-3 border-b border-yellow-200 flex justify-center gap-2">
                      <button
                        className="p-2 rounded-full bg-blue-500 text-white shadow-md hover:bg-blue-600 transition"
                        onClick={() => {
                          setEditProductIndex(i);
                          setNewProduct(p);
                          setShowProductForm(true);
                        }}
                        title="Edit"
                      >
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M11 4H4a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-7"></path><path d="M18.5 2.5a2.121 2.121 0 0 1 3 3L12 15l-4 1 1-4 9.5-9.5z"></path></svg>
                      </button>
                      <button
                        className="p-2 rounded-full bg-red-500 text-white shadow-md hover:bg-red-600 transition"
                        onClick={() => setConfirmProductDelete({ show: true, index: i })}
                        title="Delete"
                      >
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M3 6h18"></path><path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"></path></svg>
                      </button>
                      <button
                        className="p-2 rounded-full bg-yellow-500 text-white shadow-md hover:bg-yellow-600 transition"
                        onClick={() => setConfirmProductPrint({ show: true, index: i })}
                        title="Print"
                      >
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M6 9V2h12v7"></path><path d="M6 18H4a2 2 0 0 1-2-2v-5a2 2 0 0 1 2-2h16a2 2 0 0 1 2 2v5a2 2 0 0 1-2 2h-2"></path><path d="M18 14H6a2 2 0 0 0-2 2v4a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2v-4a2 2 0 0 0-2-2z"></path></svg>
                      </button>
                    </td>
                  </motion.tr>
                ))}
                {products.length === 0 && (
                  <tr>
                    <td colSpan={9} className="p-6 text-gray-500 text-center text-lg">
                      Belum ada produk. Klik 'Tambah Produk' untuk memulai.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>
        </div>
      )}

      {activeTab === 'transaction' && (
        <div className="flex flex-col flex-grow">
          <div className="flex justify-between items-center mb-6">
            <h2 className="text-2xl font-bold text-yellow-700 flex items-center gap-2">
              <span className="text-3xl">📝</span> Transaksi
            </h2>
            <button
              className="px-6 py-3 rounded-full bg-yellow-500 text-white font-semibold shadow-lg hover:bg-yellow-600 transition-transform transform hover:scale-105"
              onClick={() => {
                setShowTransactionForm(true);
                setEditTransactionIndex(null);
                setNewTransaction({ kode: "", nama: "", warna: "", size: "S", stok: "", diskon: "", admin: "", keterangan: "" });
              }}
            >
              <span className="flex items-center gap-2">
                <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M12 6v6m0 0v6m0-6h6m-6 0H6"></path></svg> Tambah Transaksi
              </span>
            </button>
          </div>

          {/* Transaction Table */}
          <div className="overflow-x-auto bg-white rounded-2xl shadow-xl flex-grow">
            <table className="w-full border-collapse">
              <thead className="bg-yellow-200 text-yellow-900 sticky top-0">
                <tr>
                  <th className="p-4 border-b-2 border-yellow-300">No.</th>
                  <th className="p-4 border-b-2 border-yellow-300">Kode</th>
                  <th className="p-4 border-b-2 border-yellow-300">Nama Produk</th>
                  <th className="p-4 border-b-2 border-yellow-300">Warna</th>
                  <th className="p-4 border-b-2 border-yellow-300">Ukuran</th>
                  <th className="p-4 border-b-2 border-yellow-300">Stok</th>
                  <th className="p-4 border-b-2 border-yellow-300">Diskon</th>
                  <th className="p-4 border-b-2 border-yellow-300">Admin</th>
                  <th className="p-4 border-b-2 border-yellow-300">Keterangan</th>
                  <th className="p-4 border-b-2 border-yellow-300">Action</th>
                </tr>
              </thead>
              <tbody>
                {transactions.map((t, i) => (
                  <motion.tr
                    key={i}
                    className="text-center even:bg-yellow-50 hover:bg-yellow-100 transition-colors"
                    initial={{ opacity: 0, y: 10 }}
                    animate={{ opacity: 1, y: 0 }}
                    transition={{ duration: 0.3, delay: i * 0.05 }}
                  >
                    <td className="p-3 border-b border-yellow-200">{i + 1}</td>
                    <td className="p-3 border-b border-yellow-200">{t.kode}</td>
                    <td className="p-3 border-b border-yellow-200">{t.nama}</td>
                    <td className="p-3 border-b border-yellow-200">{t.warna}</td>
                    <td className="p-3 border-b border-yellow-200">{t.size}</td>
                    <td className="p-3 border-b border-yellow-200">{t.stok}</td>
                    <td className="p-3 border-b border-yellow-200 font-semibold">Rp {formatNumber(t.diskon)}</td>
                    <td className="p-3 border-b border-yellow-200 font-semibold">Rp {formatNumber(t.admin)}</td>
                    <td className="p-3 border-b border-yellow-200">{t.keterangan}</td>
                    <td className="p-3 border-b border-yellow-200 flex justify-center gap-2">
                      <button
                        className="p-2 rounded-full bg-blue-500 text-white shadow-md hover:bg-blue-600 transition"
                        onClick={() => {
                          setEditTransactionIndex(i);
                          setNewTransaction(t);
                          setShowTransactionForm(true);
                        }}
                        title="Edit"
                      >
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M11 4H4a2 2 0 0 0-2 2v14a2 2 0 0 0 2 2h14a2 2 0 0 0 2-2v-7"></path><path d="M18.5 2.5a2.121 2.121 0 0 1 3 3L12 15l-4 1 1-4 9.5-9.5z"></path></svg>
                      </button>
                      <button
                        className="p-2 rounded-full bg-red-500 text-white shadow-md hover:bg-red-600 transition"
                        onClick={() => setConfirmTransactionDelete({ show: true, index: i })}
                        title="Delete"
                      >
                        <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24" xmlns="http://www.w3.org/2000/svg"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M3 6h18"></path><path d="M19 6v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6m3 0V4a2 2 0 0 1 2-2h4a2 2 0 0 1 2 2v2"></path></svg>
                      </button>
                    </td>
                  </motion.tr>
                ))}
                {transactions.length === 0 && (
                  <tr>
                    <td colSpan={10} className="p-6 text-gray-500 text-center text-lg">
                      Belum ada transaksi. Klik 'Tambah Transaksi' untuk memulai.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>
          </div>
        </div>
      )}


      {/* Footer */}
      <footer className="mt-8 text-center text-sm text-gray-500">
        <p>copyright ocikcgallery@2025</p>
      </footer>

      {/* Popup Tambah/Edit Produk */}
      {showProductForm && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50 p-4">
          <motion.div
            initial={{ scale: 0.8, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            className="bg-white rounded-2xl p-8 shadow-2xl w-full max-w-lg"
          >
            <h2 className="text-2xl font-bold text-yellow-700 mb-6">{editProductIndex !== null ? "Edit Produk" : "Tambah Produk"}</h2>
            <div className="space-y-4">
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Kode Produk</label>
                <input
                  type="text"
                  value={newProduct.kode}
                  onChange={(e) => setNewProduct({ ...newProduct, kode: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Nama Produk</label>
                <input
                  type="text"
                  value={newProduct.nama}
                  onChange={(e) => setNewProduct({ ...newProduct, nama: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Warna</label>
                <input
                  type="text"
                  value={newProduct.warna}
                  onChange={(e) => setNewProduct({ ...newProduct, warna: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Ukuran</label>
                <select
                  value={newProduct.size}
                  onChange={(e) => setNewProduct({ ...newProduct, size: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                >
                  <option value="">- Pilih -</option>
                  <option value="S">S</option>
                  <option value="L">L</option>
                  <option value="XL">XL</option>
                  <option value="ALL SIZE">ALL SIZE</option>
                </select>
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Stok</label>
                <input
                  type="text"
                  value={newProduct.stok}
                  onChange={(e) => {
                    const val = e.target.value.replace(/[^\d]/g, "");
                    setNewProduct({ ...newProduct, stok: formatNumber(val) });
                  }}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Harga</label>
                <input
                  type="text"
                  value={newProduct.harga}
                  onChange={(e) => {
                    const val = e.target.value.replace(/[^\d]/g, "");
                    setNewProduct({ ...newProduct, harga: formatNumber(val) });
                  }}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">HPP</label>
                <input
                  type="text"
                  value={newProduct.hpp}
                  onChange={(e) => {
                    const val = e.target.value.replace(/[^\d]/g, "");
                    setNewProduct({ ...newProduct, hpp: formatNumber(val) });
                  }}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
            </div>
            <div className="flex justify-end mt-6 gap-3">
              <button
                className="px-6 py-3 bg-gray-300 rounded-full font-semibold hover:bg-gray-400 transition"
                onClick={() => setShowProductForm(false)}
              >
                Batal
              </button>
              <button
                className="px-6 py-3 bg-yellow-500 text-white rounded-full font-semibold shadow-lg hover:bg-yellow-600 transition"
                onClick={handleSaveProduct}
              >
                Simpan
              </button>
            </div>
          </motion.div>
        </div>
      )}

      {/* Popup Tambah/Edit Transaksi */}
      {showTransactionForm && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50 p-4">
          <motion.div
            initial={{ scale: 0.8, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            className="bg-white rounded-2xl p-8 shadow-2xl w-full max-w-lg"
          >
            <h2 className="text-2xl font-bold text-yellow-700 mb-6">{editTransactionIndex !== null ? "Edit Transaksi" : "Tambah Transaksi"}</h2>
            <div className="space-y-4">
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Kode</label>
                <input
                  type="text"
                  value={newTransaction.kode}
                  onChange={(e) => setNewTransaction({ ...newTransaction, kode: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Nama Produk</label>
                <input
                  type="text"
                  value={newTransaction.nama}
                  onChange={(e) => setNewTransaction({ ...newTransaction, nama: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Warna</label>
                <input
                  type="text"
                  value={newTransaction.warna}
                  onChange={(e) => setNewTransaction({ ...newTransaction, warna: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Ukuran</label>
                <select
                  value={newTransaction.size}
                  onChange={(e) => setNewTransaction({ ...newTransaction, size: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                >
                  <option value="">- Pilih -</option>
                  <option value="S">S</option>
                  <option value="L">L</option>
                  <option value="XL">XL</option>
                  <option value="ALL SIZE">ALL SIZE</option>
                </select>
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Stok</label>
                <input
                  type="text"
                  value={newTransaction.stok}
                  onChange={(e) => {
                    const val = e.target.value.replace(/[^\d]/g, "");
                    setNewTransaction({ ...newTransaction, stok: formatNumber(val) });
                  }}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Diskon</label>
                <input
                  type="text"
                  value={newTransaction.diskon}
                  onChange={(e) => {
                    const val = e.target.value.replace(/[^\d]/g, "");
                    setNewTransaction({ ...newTransaction, diskon: formatNumber(val) });
                  }}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Admin</label>
                <input
                  type="text"
                  value={newTransaction.admin}
                  onChange={(e) => {
                    const val = e.target.value.replace(/[^\d]/g, "");
                    setNewTransaction({ ...newTransaction, admin: formatNumber(val) });
                  }}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
              <div className="flex items-center gap-4">
                <label className="text-gray-700 font-medium w-32">Keterangan</label>
                <input
                  type="text"
                  value={newTransaction.keterangan}
                  onChange={(e) => setNewTransaction({ ...newTransaction, keterangan: e.target.value })}
                  className="flex-grow p-3 border border-gray-300 rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-500"
                />
              </div>
            </div>
            <div className="flex justify-end mt-6 gap-3">
              <button
                className="px-6 py-3 bg-gray-300 rounded-full font-semibold hover:bg-gray-400 transition"
                onClick={() => setShowTransactionForm(false)}
              >
                Batal
              </button>
              <button
                className="px-6 py-3 bg-yellow-500 text-white rounded-full font-semibold shadow-lg hover:bg-yellow-600 transition"
                onClick={handleSaveTransaction}
              >
                Simpan
              </button>
            </div>
          </motion.div>
        </div>
      )}


      {/* Popup Konfirmasi Print Produk */}
      {confirmProductPrint.show && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50 p-4">
          <motion.div
            initial={{ scale: 0.8, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            className="bg-white rounded-2xl p-8 shadow-2xl w-full max-w-sm text-center"
          >
            <h2 className="text-xl font-bold text-yellow-700 mb-4">Konfirmasi</h2>
            <p className="text-gray-700 mb-6">Apakah Anda yakin ingin mencetak label produk ini?</p>
            <div className="flex justify-center gap-3">
              <button
                className="px-6 py-3 bg-gray-300 rounded-full font-semibold hover:bg-gray-400 transition"
                onClick={() => setConfirmProductPrint({ show: false, index: null })}
              >
                Tidak
              </button>
              <button
                className="px-6 py-3 bg-yellow-500 text-white rounded-full font-semibold shadow-lg hover:bg-yellow-600 transition"
                onClick={handlePrintProduct}
              >
                Ya
              </button>
            </div>
          </motion.div>
        </div>
      )}

      {/* Popup Konfirmasi Hapus Produk */}
      {confirmProductDelete.show && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50 p-4">
          <motion.div
            initial={{ scale: 0.8, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            className="bg-white rounded-2xl p-8 shadow-2xl w-full max-w-sm text-center"
          >
            <h2 className="text-xl font-bold text-red-600 mb-4">Konfirmasi Hapus</h2>
            <p className="text-gray-700 mb-6">Apakah Anda yakin ingin menghapus produk ini?</p>
            <div className="flex justify-center gap-3">
              <button
                className="px-6 py-3 bg-gray-300 rounded-full font-semibold hover:bg-gray-400 transition"
                onClick={() => setConfirmProductDelete({ show: false, index: null })}
              >
                Tidak
              </button>
              <button
                className="px-6 py-3 bg-red-500 text-white rounded-full font-semibold shadow-lg hover:bg-red-600 transition"
                onClick={handleDeleteProduct}
              >
                Ya
              </button>
            </div>
          </motion.div>
        </div>
      )}

      {/* Popup Konfirmasi Hapus Transaksi */}
      {confirmTransactionDelete.show && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50 p-4">
          <motion.div
            initial={{ scale: 0.8, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            className="bg-white rounded-2xl p-8 shadow-2xl w-full max-w-sm text-center"
          >
            <h2 className="text-xl font-bold text-red-600 mb-4">Konfirmasi Hapus</h2>
            <p className="text-gray-700 mb-6">Apakah Anda yakin ingin menghapus transaksi ini?</p>
            <div className="flex justify-center gap-3">
              <button
                className="px-6 py-3 bg-gray-300 rounded-full font-semibold hover:bg-gray-400 transition"
                onClick={() => setConfirmTransactionDelete({ show: false, index: null })}
              >
                Tidak
              </button>
              <button
                className="px-6 py-3 bg-red-500 text-white rounded-full font-semibold shadow-lg hover:bg-red-600 transition"
                onClick={handleDeleteTransaction}
              >
                Ya
              </button>
            </div>
          </motion.div>
        </div>
      )}

      {/* Info Message Modal */}
      {infoMessage.show && (
        <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50 p-4">
          <motion.div
            initial={{ scale: 0.8, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            className="bg-white rounded-2xl p-8 shadow-2xl w-full max-w-sm text-center"
          >
            <h2 className="text-xl font-bold text-yellow-700 mb-4">Pemberitahuan</h2>
            <p className="text-gray-700 mb-6">{infoMessage.message}</p>
            <div className="flex justify-center">
              <button
                className="px-6 py-3 bg-gray-300 rounded-full font-semibold hover:bg-gray-400 transition"
                onClick={() => setInfoMessage({ show: false, message: "" })}
              >
                OK
              </button>
            </div>
          </motion.div>
        </div>
      )}
    </div>
  );
}
